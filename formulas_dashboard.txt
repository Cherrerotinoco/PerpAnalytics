; FÓRMULAS PARA DASHBOARD DE TRADING
; ═══════════════════════════════════════════════════════════════════════════════
;
; Instrucciones:
;   1. Copia las fórmulas que necesites directamente en las celdas de tu Excel.
;   2. Todas las fórmulas usan referencias estructuradas a la tabla "Data"
;      (la que está en tu hoja "trading_dashboard").
;   3. El separador de argumentos es ; (tu configuración regional usa coma decimal).
;   4. Las funciones están en INGLÉS (SUM, IF, COUNTIF, etc.).
;   5. Las columnas auxiliares (Peak, DD, Streak) van en tu tabla Data.
;
; ═══════════════════════════════════════════════════════════════════════════════


; ─────────────────────────────────────────────────────────────────────────────
; 1. MÉTRICAS BÁSICAS (reemplazan los GETPIVOTDATA)
; ─────────────────────────────────────────────────────────────────────────────

; Resultado del mes (PnL total)
=SUM(Data[Result])

; Operaciones Totales
=COUNTA(Data[Result])

; Operaciones ganadas
=SUMPRODUCT((Data[Result]>0)*1)

; Operaciones perdidas
=SUMPRODUCT((Data[Result]<0)*1)

; % Win Rate
=SUMPRODUCT((Data[Result]>0)*1)/COUNTA(Data[Result])

; % Loss Rate
=SUMPRODUCT((Data[Result]<0)*1)/COUNTA(Data[Result])

; Máximo ganado (mejor trade)
=MAX(Data[Result])

; Máximo perdido (peor trade)
=MIN(Data[Result])

; Promedio general
=AVERAGE(Data[Result])

; Promedio de trades ganadores
=AVERAGEIF(Data[Result];">0")

; Promedio de trades perdedores
=AVERAGEIF(Data[Result];"<0")

; Total fees pagadas
=SUM(Data[Fees (USD)])


; ─────────────────────────────────────────────────────────────────────────────
; 2. PROFIT FACTOR & RISK/REWARD
; ─────────────────────────────────────────────────────────────────────────────

; Profit Factor (ganancia bruta / pérdida bruta)
; Un PF > 1 = rentable. > 2 = muy bueno.
=SUMPRODUCT((Data[Result]>0)*Data[Result])/ABS(SUMPRODUCT((Data[Result]<0)*Data[Result]))

; Risk/Reward Ratio (avg win / avg loss)
=AVERAGEIF(Data[Result];">0")/ABS(AVERAGEIF(Data[Result];"<0"))

; Expectancia por trade (cuánto esperas ganar por operación)
=(SUMPRODUCT((Data[Result]>0)*1)/COUNTA(Data[Result]))*AVERAGEIF(Data[Result];">0")+(SUMPRODUCT((Data[Result]<0)*1)/COUNTA(Data[Result]))*AVERAGEIF(Data[Result];"<0")


; ─────────────────────────────────────────────────────────────────────────────
; 3. DRAWDOWN & MAX DRAWDOWN
; ─────────────────────────────────────────────────────────────────────────────
;
; Necesitas 2 columnas auxiliares en tu tabla Data:
;   - "Peak"  (columna R o donde quieras)
;   - "DD"    (columna S o donde quieras)
;
; Añádelas como columnas nuevas a la tabla.

; Columna "Peak" — máximo del cumulative result hasta esa fila
; (Fila 2, primera fila de datos):
=MAX(C$2:[@[Cumulative result]])

; Columna "DD" — drawdown en cada fila (será ≤ 0)
=[@[Cumulative result]]-[@Peak]

; ── Métricas de drawdown para el dashboard: ──

; Max Drawdown (valor más negativo)
=MIN(Data[DD])

; Max Drawdown % (si quieres relativo al peak)
=MIN(Data[DD]/IF(Data[Peak]<>0;Data[Peak];1))

; Drawdown actual (último DD)
=INDEX(Data[DD];COUNTA(Data[DD]))


; ─────────────────────────────────────────────────────────────────────────────
; 4. RACHAS (CONSECUTIVE WINS / LOSSES)
; ─────────────────────────────────────────────────────────────────────────────
;
; Añade columna "Streak" a tu tabla Data.
; Positivo = racha ganadora. Negativo = racha perdedora.

; Columna "Streak" — primera fila de datos (fila 2):
=IF([@Result]>0;1;-1)

; Columna "Streak" — a partir de la segunda fila de datos (fila 3 en adelante):
; Copia esta fórmula. Asume que la celda de arriba es la fila anterior de Streak.
; Si usas tabla estructurada, necesitas referencia mixta:
=IF([@Result]>0;IF(INDIRECT("R"&ROW()-1&"C"&COLUMN();FALSE)>0;INDIRECT("R"&ROW()-1&"C"&COLUMN();FALSE)+1;1);IF(INDIRECT("R"&ROW()-1&"C"&COLUMN();FALSE)<0;INDIRECT("R"&ROW()-1&"C"&COLUMN();FALSE)-1;-1))

; ── ALTERNATIVA MÁS SIMPLE para Streak (usando celda directa, ej T2, T3...): ──
; En T2: =IF(B2>0;1;-1)
; En T3: =IF(B3>0;IF(T2>0;T2+1;1);IF(T2<0;T2-1;-1))
; Copiar T3 hacia abajo.

; ── Métricas de rachas para el dashboard: ──

; Racha ganadora más larga
=MAX(Data[Streak])

; Racha perdedora más larga (valor absoluto)
=ABS(MIN(Data[Streak]))

; Racha actual
=INDEX(Data[Streak];COUNTA(Data[Streak]))


; ─────────────────────────────────────────────────────────────────────────────
; 5. ANÁLISIS POR DÍA DE LA SEMANA
; ─────────────────────────────────────────────────────────────────────────────

; PnL por día
=SUMIF(Data[Days];"lunes";Data[Result])
=SUMIF(Data[Days];"martes";Data[Result])
=SUMIF(Data[Days];"miércoles";Data[Result])
=SUMIF(Data[Days];"jueves";Data[Result])
=SUMIF(Data[Days];"viernes";Data[Result])
=SUMIF(Data[Days];"sábado";Data[Result])
=SUMIF(Data[Days];"domingo";Data[Result])

; Número de trades por día
=COUNTIF(Data[Days];"lunes")
=COUNTIF(Data[Days];"martes")
=COUNTIF(Data[Days];"miércoles")
=COUNTIF(Data[Days];"jueves")
=COUNTIF(Data[Days];"viernes")
=COUNTIF(Data[Days];"sábado")
=COUNTIF(Data[Days];"domingo")

; Win rate por día (ej. lunes)
=SUMPRODUCT((Data[Days]="lunes")*(Data[Result]>0))/COUNTIF(Data[Days];"lunes")

; Mejor día (el que más PnL acumula)
=INDEX({"lunes";"martes";"miércoles";"jueves";"viernes";"sábado";"domingo"};MATCH(MAX(SUMIF(Data[Days];{"lunes";"martes";"miércoles";"jueves";"viernes";"sábado";"domingo"};Data[Result]));SUMIF(Data[Days];{"lunes";"martes";"miércoles";"jueves";"viernes";"sábado";"domingo"};Data[Result]);0))

; Peor día
=INDEX({"lunes";"martes";"miércoles";"jueves";"viernes";"sábado";"domingo"};MATCH(MIN(SUMIF(Data[Days];{"lunes";"martes";"miércoles";"jueves";"viernes";"sábado";"domingo"};Data[Result]));SUMIF(Data[Days];{"lunes";"martes";"miércoles";"jueves";"viernes";"sábado";"domingo"};Data[Result]);0))


; ─────────────────────────────────────────────────────────────────────────────
; 6. MÉTRICAS POR SIDE (LONG vs SHORT)
; ─────────────────────────────────────────────────────────────────────────────

; PnL total long
=SUMIF(Data[Side];"long";Data[Result])

; PnL total short
=SUMIF(Data[Side];"short";Data[Result])

; Win rate long
=SUMPRODUCT((Data[Side]="long")*(Data[Result]>0))/COUNTIF(Data[Side];"long")

; Win rate short
=SUMPRODUCT((Data[Side]="short")*(Data[Result]>0))/COUNTIF(Data[Side];"short")

; Trades long
=COUNTIF(Data[Side];"long")

; Trades short
=COUNTIF(Data[Side];"short")


; ─────────────────────────────────────────────────────────────────────────────
; 7. MÉTRICAS DE TIEMPO
; ─────────────────────────────────────────────────────────────────────────────

; Duración promedio (en horas)
=AVERAGE(Data[Hours])

; Duración promedio trades ganadores (horas)
=AVERAGEIFS(Data[Hours];Data[Result];">0")

; Duración promedio trades perdedores (horas)
=AVERAGEIFS(Data[Hours];Data[Result];"<0")

; Trade más largo (horas)
=MAX(Data[Hours])

; Trade más corto (horas)
=MIN(Data[Hours])
